cmake_minimum_required(VERSION 3.22)
project(pydaily_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory(llama.cpp)
add_library(pydaily_llm SHARED tutor.cpp)
target_include_directories(pydaily_llm PRIVATE llama.cpp/include)
target_link_libraries(pydaily_llm PRIVATE llama)
if(ANDROID)
 target_link_libraries(pydaily_llm PRIVATE android log)
else()
 target_include_directories(pydaily_llm PRIVATE "${PYDAILY_JNI_INCLUDE}")
endif()
target_link_options(pydaily_llm PRIVATE "-Wl,-z,max-page-size=16384")
